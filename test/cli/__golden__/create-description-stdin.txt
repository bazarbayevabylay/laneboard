# create-description-stdin
$ laneboard create 'From stdin' --status todo --description-file - --as a
exit 0
--- stdout
LB-1 created (rev 1, todo, unassigned)
--- stderr


$ laneboard create 'From stdin' --status todo --description-file - --as a --json
exit 0
--- stdout

--- stderr
